function getSpreadsheet() {
  return SpreadsheetApp.openById(SPREADSHEET_ID);
}

function getGuestSheet() {
  return getSpreadsheet().getSheetByName(GUEST_SHEET);
}

function getRsvpSheet() {
  return getSpreadsheet().getSheetByName(RSVP_SHEET);
}

function json(data) {
  return ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

function findGuest(id) {

  const data = getGuestSheet().getDataRange().getValues();

  for (let i = 1; i < data.length; i++) {

    if (String(data[i][0]).trim() === String(id).trim()) {

      return {
        row: i + 1,
        id: data[i][0],
        nama: data[i][1]
      };

    }

  }

  return null;

}

function findRsvp(id){

  const rows = getRsvpSheet().getDataRange().getValues();

  for(let i=1;i<rows.length;i++){

      if(String(rows[i][0]).trim() === String(id).trim()){

          return{
              row:i+1,
              id:rows[i][0]
          }

      }

  }

  return null;

}
